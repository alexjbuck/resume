\colorSection{Projects}

    \itemcolor{Pathvein:} A library for completing expressive and

    \itemcolor{Badmax:} A tool for building Air Plan documents from

    Post Mission Analysis files from the MH-60R.
    \itemcolor{WaveBreaker:} A tool for planning weapon impact points
    to cover the area of uncertainty from an unobserved target.
\end{enumerate}
